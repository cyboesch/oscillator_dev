% Six continuous visible oscillators in a slightly oblique spring lattice.
% Coordinates are prepared separately so every hidden edge is drawn behind nodes.
\newcommand{\latticecoords}[3]{%
  \foreach \r in {0,1}{\foreach \c in {0,1,2}{
    \coordinate (#3\r\c) at (#1+11*\c-4*\r,#2-9*\r);
  }}
  \fill[bluepale] (#300) -- (#302) -- (#312) -- (#310) -- cycle;
}
\newcommand{\latticedraw}[1]{%
  \foreach \r in {0,1}{\foreach \c [evaluate=\c as \next using int(\c+1)] in {0,1}{
    \draw[spring] (#1\r\c) -- (#1\r\next);
  }}
  \foreach \c in {0,1,2}{\draw[spring] (#10\c) -- (#11\c);}
  \foreach \r in {0,1}{\foreach \c in {0,1,2}{
    \node[visible] at (#1\r\c) {};
    \fill[blue] (#1\r\c) circle (.52mm);
  }}
}

% Deterministic 12 x 12 illustration, not a model sample or measured trajectory.
% The third argument is a graphic blend fraction, not physical/diffusion time.
\newcommand{\thumbnail}[4]{%
  \begin{scope}[shift={(#1,#2)},x=#4mm,y=#4mm]
    \pgfmathsetseed{31415}
    \foreach \row in {0,...,11}{\foreach \col in {0,...,11}{
      % A hand-drawn '3': two open arcs and a short central bar.
      \pgfmathtruncatemacro{\digit}{
        ((\row==1 || \row==10) && \col>=4 && \col<=7) ||
        ((\row==2 || \row==9) && (\col==3 || \col==8)) ||
        ((\row==3 || \row==4 || \row==7 || \row==8) && \col==8) ||
        ((\row==5 || \row==6) && \col>=5 && \col<=7)}
      \pgfmathrandominteger{\noise}{0}{100}
      \pgfmathtruncatemacro{\shade}{round((1-#3)*\noise+#3*94*\digit)}
      \fill[white!\shade!black] (\col,-\row) rectangle ++(1,-1);
    }}
    \draw[rule,line width=.45pt] (0,0) rectangle (12,-12);
  \end{scope}
}

% Shared schematic schedule icon: the identical curves appear in b and c.
\newcommand{\scheduleicon}[2]{%
  \begin{scope}[shift={(#1,#2)}]
    \draw[orange,line width=1bp] (0,0) .. controls (2,4.9) and (5,-1.5) .. (8,3.7);
    \draw[violet,line width=1bp] (0,3.7) .. controls (3,-.1) and (5,4.9) .. (8,.1);
  \end{scope}
}

% Fit each badge to its text plus the shared icon; keep padding identical.
\newcommand{\schedulebadge}[5]{%
  \node[anchor=#4,fill=orangepale,rounded corners=1.4mm,
    inner xsep=2.5mm,inner ysep=1.5mm,minimum height=7.2mm] (#5) at (#1,#2)
    {\hspace*{10.5mm}#3};
  \begin{scope}[shift={($(#5.west)+(2.5,-1.85)$)}]
    \scheduleicon{0}{0}
  \end{scope}
}

% Actual 28 x 28 pixels, on a shared fixed intensity scale across all times.
\newcommand{\sampleframe}[3]{%
  \begin{scope}[shift={(#1,#2)},x=.7857142857mm,y=.7857142857mm]
    \input{conceptual_frame_#3.tex}
    \draw[rule,line width=.45pt] (0,0) rectangle (28,-28);
  \end{scope}
}
